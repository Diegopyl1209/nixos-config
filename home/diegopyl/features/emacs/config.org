* Appeareance
** Disable Menubar, Toolbars and Scrollbar
#+begin_src elisp

  (menu-bar-mode 0)
  (tool-bar-mode 0)
  (tooltip-mode 0)
  (scroll-bar-mode 0)
  (set-fringe-mode 0)
#+end_src

** Display Line numbers
#+begin_src elisp
  (setq display-line-numbers-type 'relative)
  (global-display-line-numbers-mode t) 
  (global-visual-line-mode t)
#+end_src
*** Make line numbers dont appear in certain modes
#+begin_src elisp
(dolist (mode '(org-mode-hook
                    term-mode-hook
                    vterm-mode-hook
                    shell-mode-hook
                    treemacs-mode-hook))
      (add-hook mode (lambda() (display-line-numbers-mode 0))))
#+end_src

** Theme and Modeline
#+begin_src elisp
  (modify-all-frames-parameters '((internal-border-width . 32))) ; padding between emacs and frame

  (doom-modeline-mode 1)

  (load-theme 'doom-gruvbox t)

  (add-to-list 'default-frame-alist '(background-color . "#202020"))

#+end_src

* General Keybindings
#+begin_src elisp
  (general-define-key
    "<escape>" 'keyboard-escape-quit
   "M-o" 'ace-window
   "\C-s" 'swiper
   "C-c C-r" 'ivy-resume
   "<f6>" 'ivy-resume
   "M-x" 'counsel-M-x
   "C-x C-f" 'counsel-find-file
   "<f1> f" 'counsel-describe-function
   "<f1> v" 'counsel-describe-variable
   "<f1> o" 'counsel-describe-symbol
   "<f1> l" 'counsel-find-library
   "<f2> i" 'counsel-info-lookup-symbol
   "<f2> u" 'counsel-unicode-char 
   )

  ; make C-c dont work in insert mode to avoid issues with ctrl-c in vterm or similars
  (general-create-definer my-leader-def
    :states '(normal visual emacs)
    :prefix "C-c")

  (my-leader-def
    "b" '(:ignore t)
    "bb" 'switch-to-buffer
    "bk" 'kill-buffer
    "bn" 'next-buffer
    "bp" 'previous-buffer
    "br" 'revert-buffer
    "g" 'counsel-git
    "j" 'counsel-git-grep
    "l" 'counsel-locate
    "a" 'org-agenda
    "c" 'org-capture  
   )

  (general-define-key
   :states '(normal visual)
   :keymaps 'org-mode-map
   "gj" 'org-next-visible-heading
   "gk" 'org-previous-visible-heading)

#+end_src
